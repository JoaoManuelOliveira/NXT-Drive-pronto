	<script>
		function switchTab(tab) {
			const formLogin = document.getElementById('form-login');
			const formCadastro = document.getElementById('form-cadastro');
			const tabs = document.querySelectorAll('.tab-btn');

			tabs.forEach(btn => btn.classList.remove('active'));

			if (tab === 'login') {
				formLogin.classList.remove('hidden');
				formCadastro.classList.add('hidden');
				tabs[0].classList.add('active');
			} else {
				formCadastro.classList.remove('hidden');
				formLogin.classList.add('hidden');
				tabs[1].classList.add('active');
			}
		}

		window.addEventListener('DOMContentLoaded', () => {
			const urlParams = new URLSearchParams(window.location.search);

			// Se a URL tiver ?error, garante que abre a aba de login
			if (urlParams.has('error')) {
				switchTab('login');
			}
			// Se veio de redirecionamento de erro de cadastro
			else if (urlParams.get('aba') === 'cadastro' || window.location.pathname.includes('/cadastro')) {
				switchTab('cadastro');
			}
		});

		// Oculta todas as mensagens (.msg-box) após 4 segundos
		setTimeout(() => {
			const msgs = document.querySelectorAll('.msg-box');
			msgs.forEach(msg => msg.style.display = 'none');
		}, 4000);
	</script>

		function consultarCep(cepLimpo) {
			fetch(`/api/cep/${cepLimpo}`)
				.then(response => {
					if (!response.ok) throw new Error("Erro na requisição");
					return response.json();
				})
				.then(data => {
					if (data && !data.erro) {
						document.getElementById("logradouroInput").value = data.logradouro || '';
						document.getElementById("bairroInput").value = data.bairro || '';
						document.getElementById("localidadeInput").value = data.localidade || '';
						document.getElementById("ufInput").value = data.uf || '';
						document.getElementById("numeroInput").focus();
					} else {
						alert("CEP não encontrado.");
					}
				})
				.catch(error => {
					console.error("Erro ao consultar CEP na API local:", error);
					alert("Erro ao realizar consulta de CEP.");
				});
		}

		async function salvarPerfil(event) {
			event.preventDefault();

			const nome = document.getElementById("perfilNome").value;
			const cep = document.getElementById("cepInput").value.trim();
			const logradouro = document.getElementById("logradouroInput").value.trim();
			const numero = document.getElementById("numeroInput").value.trim();
			const bairro = document.getElementById("bairroInput").value.trim();
			const localidade = document.getElementById("localidadeInput").value.trim();
			const uf = document.getElementById("ufInput").value.trim();
			const complemento = document.getElementById("complementoInput").value.trim();

			if (!cep || !logradouro || !numero || !bairro || !localidade || !uf) {
				alert("❌ Por favor, preencha todos os campos obrigatórios do endereço (CEP, Logradouro, Número, Bairro, Cidade e UF).");
				return;
			}

			const btn = document.getElementById("btnSalvarPerfil");
			btn.disabled = true;
			btn.innerText = "SALVANDO...";

			try {
				const res = await fetch('/api/v1/usuarios/me/perfil', {
					method: 'PUT',
					headers: {'Content-Type': 'application/json'},
					body: JSON.stringify({
						nome,
						cep,
						logradouro,
						numero,
						bairro,
						localidade,
						uf,
						complemento
					})
				});

				const data = await res.json().catch(() => ({}));

				if (!res.ok) {
					throw new Error(data.erro || "Falha ao salvar dados de perfil.");

				usuarioTemEndereco = true;

			} catch (err) {

				alert("❌ " + err.message);
			} finally {

				btn.innerText = "SALVAR ALTERAÇÕES";
			}

		async function confirmarAlteracaoSenha(event) {
			event.preventDefault();
			const senhaAtual = document.getElementById("senhaAtualInput").value;
			const novaSenha = document.getElementById("novaSenhaInput").value;
			const confirmarSenha = document.getElementById("confirmarNovaSenhaInput").value;
			const btn = document.getElementById("btnSalvarSenha");

			if (novaSenha !== confirmarSenha) {
				alert("❌ A nova senha e a confirmação de senha não coincidem!");
				return;
			}

			if (novaSenha.length < 6) {
				alert("❌ A nova senha deve possuir no mínimo 6 caracteres!");
				return;
			}

			btn.disabled = true;
			btn.innerText = "ALTERANDO SENHA...";

			try {
				const res = await fetch('/api/v1/usuarios/alterar-senha', {
					method: 'PATCH',
					headers: {'Content-Type': 'application/json'},
					body: JSON.stringify({
						senhaAtual: senhaAtual,
						novaSenha: novaSenha,
						confirmacaoSenha: confirmarSenha
					})
				});

				const data = await res.json().catch(() => ({}));

				if (!res.ok) {
					throw new Error(data.erro || "Falha ao alterar senha.");
				}

				alert("🔒 " + (data.mensagem || "Senha alterada com sucesso!"));

				btn.disabled = false;

		function mostrarNotificacao(nomeProduto) {
			const container = document.getElementById("toastContainer");
			const toast = document.createElement("div");
			toast.className = "toast";
			toast.innerHTML = `
				<span class="toast-icon">✓</span>
				<span><strong>${nomeProduto}</strong> adicionado ao carrinho!</span>
			`;
			container.appendChild(toast);

			setTimeout(() => {toast.remove();}, 3000);
		}

		function atualizarCarrinho() {
			const contador = document.getElementById("contadorCarrinho");
			contador.innerText = carrinho.length;

			const lista = document.getElementById("listaCarrinho");
			const total = document.getElementById("valorTotal");

			if (carrinho.length === 0) {
				lista.innerHTML = '<p class="carrinho-vazio">Seu carrinho está vazio.</p>';
				total.innerText = "R$ 0,00";
				return;
			}

			lista.innerHTML = "";
			let valorTotal = 0;

			carrinho.forEach((produto, index) => {
				valorTotal += produto.valor;
				const item = document.createElement("div");
				item.className = "carrinho-item";
				item.innerHTML = `
					<div>
						<strong>${produto.nome}</strong>
						<span>R$ ${produto.valor.toFixed(2).replace(".", ",")}</span>
					</div>
					<button onclick="removerCarrinho(${index})">×</button>
				`;
				lista.appendChild(item);
			});

			total.innerText = "R$ " + valorTotal.toFixed(2).replace(".", ",");
		}

		async function obterMercadoPagoPublicKey() {
			if (mpPublicKey) return mpPublicKey;

		function abrirCheckout() {

			const total = carrinho.reduce((acc, item) => acc + item.valor, 0);

		async function iniciarCheckoutPix(total) {
			const pixCarregando = document.getElementById("pixCarregando");
			const pixDetalhes = document.getElementById("pixDetalhes");
			const pixStatusBadge = document.getElementById("pixStatusBadge");

			pixCarregando.style.display = "block";
			pixDetalhes.style.display = "none";
			pixStatusBadge.innerHTML = 'Aguardando pagamento... ⏳';
			pixStatusBadge.style.color = '#3da9ff';
			pixStatusBadge.style.borderColor = '#3da9ff';

			const emailEl = document.getElementById("usuarioEmailLogado");
			const email = (emailEl && emailEl.value) ? emailEl.value : "cliente@nxtdrive.com.br";

			try {
				const nomeCliente = document.getElementById("perfilNome")?.value || document.getElementById("usuarioNomeLogado")?.value || "Cliente NXT Drive";
				const rua = document.getElementById("logradouroInput")?.value || "";
				const numero = document.getElementById("numeroInput")?.value || "";
				const complemento = document.getElementById("complementoInput")?.value || "";
				const bairro = document.getElementById("bairroInput")?.value || "";
				const cidade = document.getElementById("localidadeInput")?.value || "";
				const uf = document.getElementById("ufInput")?.value || "";
				const cep = document.getElementById("cepInput")?.value || "";
				const itensDesc = carrinho.map(i => i.nome).join(" + ");

				const res = await fetch('/api/v1/pagamentos/pix', {
					method: 'POST',
					headers: {'Content-Type': 'application/json'},
					body: JSON.stringify({
						valor: total,
						email: email,
						descricao: "Compra Loja NXT Drive (" + carrinho.length + " itens)",
						nomeCliente: nomeCliente,
						rua: rua,
						numero: numero,
						complemento: complemento,
						bairro: bairro,
						cidade: cidade,
						uf: uf,
						cep: cep,
						itensDescricao: itensDesc
					})
				});

				if (!res.ok) throw new Error("Falha ao comunicar com o servidor");

				const data = await res.json();
				pedidoIdAtual = data.pedidoId;

				// Exibe QR Code em Base64 recebido da API do Mercado Pago
				if (data.qrCodeBase64) {
					document.getElementById("qrCodeImg").src = "data:image/png;base64," + data.qrCodeBase64;
				} else if (data.qrCode) {
					document.getElementById("qrCodeImg").src = `https://api.qrserver.com/v1/create-qr-code/?size=250x250&data=${encodeURIComponent(data.qrCode)}`;
				}

				document.getElementById("pixCopiaColaInput").value = data.qrCode || "";

				pixCarregando.style.display = "none";
				pixDetalhes.style.display = "block";

				// Inicia verificação periódica de status (polling) utilizando o UUID do pedido
				iniciarPollingStatus(pedidoIdAtual);

			} catch (err) {
				console.error("Erro ao gerar Pix:", err);
				pixCarregando.innerHTML = `<span style="color:#ff5252">❌ Erro ao gerar Pix no Mercado Pago: ${err.message}</span>`;
			}
		}

		function copiarPixCode() {
			const codigoPix = document.getElementById("pixCopiaColaInput").value;
			if (!codigoPix) return;

			navigator.clipboard.writeText(codigoPix).then(() => {
				const btnCopiar = document.getElementById("btnCopiarPix");
				btnCopiar.innerText = "✓ CÓDIGO COPIADO COM SUCESSO!";
				setTimeout(() => {
					btnCopiar.innerText = "📋 COPIAR CÓDIGO PIX";
				}, 4000);
			}).catch(err => {
				alert("Copie o código Pix manualmente:\n\n" + codigoPix);
			});
		}

		function iniciarPollingStatus(pedidoId) {
			pararPollingStatus();
			if (!pedidoId) return;

			pollingInterval = setInterval(async () => {
				try {
					const res = await fetch(`/api/v1/pagamentos/status/${pedidoId}`);
					if (res.ok) {
						const data = await res.json();
						const status = (data.status || '').toLowerCase();

						if (status === 'approved' || status === 'pago') {
							pararPollingStatus();
							const pixStatusBadge = document.getElementById("pixStatusBadge");
							if (pixStatusBadge) {
								pixStatusBadge.innerHTML = '✅ PAGAMENTO CONFIRMADO COM SUCESSO!';
								pixStatusBadge.style.color = '#00e676';
								pixStatusBadge.style.borderColor = '#00e676';
							}
							setTimeout(() => {
								concluirPedido("🎉 Pagamento Pix aprovado no Mercado Pago! Pedido confirmado.");
							}, 1500);
						} else if (status === 'rejected' || status === 'cancelled') {
							pararPollingStatus();
							const pixStatusBadge = document.getElementById("pixStatusBadge");
							if (pixStatusBadge) {
								pixStatusBadge.innerHTML = '❌ Pagamento rejeitado ou cancelado.';
								pixStatusBadge.style.color = '#ff5252';
								pixStatusBadge.style.borderColor = '#ff5252';
							}
						}
					}
				} catch (e) {
					console.warn("Polling de status falhou:", e);
				}
			}, 3000);
		}

		function pararPollingStatus() {
			if (pollingInterval) {
				clearInterval(pollingInterval);
				pollingInterval = null;
			}
		}

		async function processarPagamentoCartaoFrontend(event) {
			event.preventDefault();

			const btn = document.getElementById("btnPagarCartao");
			btn.disabled = true;
			btn.innerText = "PROCESSANDO PAGAMENTO...";

			const numero = document.getElementById("cartaoNumero").value.replace(/\s+/g, '');
			const nome = document.getElementById("cartaoNome").value.trim();
			const validade = document.getElementById("cartaoValidade").value.trim();
			const cvv = document.getElementById("cartaoCvv").value.trim();
			const cpf = document.getElementById("cartaoCpf").value.replace(/\D/g, '');
			const parcelas = parseInt(document.getElementById("cartaoParcelas").value, 10) || 1;

			const partesValidade = validade.split('/');
			const mes = partesValidade[0];
			let ano = partesValidade[1] || '';
			if (ano.length === 2) ano = "20" + ano;

			const total = carrinho.reduce((acc, item) => acc + item.valor, 0);
			const emailEl = document.getElementById("usuarioEmailLogado");
			const email = (emailEl && emailEl.value) ? emailEl.value : "cliente@nxtdrive.com.br";

			try {
				const publicKey = await obterMercadoPagoPublicKey();

				// 1. Tokenização direta na API pública do Mercado Pago (dados nunca passam abertos pelo backend)
				const tokenRes = await fetch(`https://api.mercadopago.com/v1/card_tokens?public_key=${encodeURIComponent(publicKey)}`, {
					method: 'POST',
					headers: {'Content-Type': 'application/json'},
					body: JSON.stringify({
						card_number: numero,
						cardholder: {
							name: nome,
							identification: {
								type: 'CPF',
								number: cpf
							}
						},
						security_code: cvv,
						expiration_month: parseInt(mes, 10),
						expiration_year: parseInt(ano, 10)
					})
				});

				if (!tokenRes.ok) {
					const errorData = await tokenRes.json();
					throw new Error(errorData.message || "Erro na validação do cartão pelo Mercado Pago");
				}

				const tokenData = await tokenRes.json();
				const cardToken = tokenData.id;

				const rua = document.getElementById("logradouroInput")?.value || "";
				const numero = document.getElementById("numeroInput")?.value || "";
				const complemento = document.getElementById("complementoInput")?.value || "";
				const bairro = document.getElementById("bairroInput")?.value || "";
				const cidade = document.getElementById("localidadeInput")?.value || "";
				const uf = document.getElementById("ufInput")?.value || "";
				const cep = document.getElementById("cepInput")?.value || "";
				const itensDesc = carrinho.map(i => i.nome).join(" + ");

				// 2. Envio seguro apenas do token gerado para o backend Spring Boot
				const payRes = await fetch('/api/v1/pagamentos/cartao', {
					method: 'POST',
					headers: {'Content-Type': 'application/json'},
					body: JSON.stringify({
						token: cardToken,
						parcelas: parcelas,
						email: email,
						valor: total,
						descricao: "Compra Cartão NXT Drive (" + carrinho.length + " itens)",
						nomeCliente: nome,
						cpfCliente: cpf,
						rua: rua,
						numero: numero,
						complemento: complemento,
						bairro: bairro,
						cidade: cidade,
						uf: uf,
						cep: cep,
						itensDescricao: itensDesc
					})
				});

				if (!payRes.ok) throw new Error("Falha ao processar pagamento com cartão no backend");

				const payData = await payRes.json();

				const statusLower = (payData.status || '').toLowerCase();
				if (statusLower === 'approved' || statusLower === 'pago') {
					concluirPedido("💳 Pagamento com Cartão Aprovado com Sucesso!");
				} else {
					alert(`Status do Pagamento: ${payData.status} - ${payData.mensagem || 'Aguarde processamento.'}`);
					concluirPedido("Pedido registrado. Status: " + payData.status);
				}

			} catch (err) {
				console.error("Erro checkout cartão:", err);
				alert("❌ Falha no pagamento: " + err.message);
			} finally {
				btn.disabled = false;
				btn.innerText = "CONFIRMAR PAGAMENTO COM CARTÃO";
			}
		}

		async function carregarPedidosAdmin(manual = false) {
			const refreshIcon = document.getElementById("refreshIcon");
			if (manual && refreshIcon) {
				refreshIcon.style.display = "inline-block";
				refreshIcon.style.animation = "spin 0.8s linear infinite";
			}

			try {
				const params = new URLSearchParams();
				if (statusSelecionado && statusSelecionado !== 'TODOS') {
					params.append('status', statusSelecionado);
				}
				const clienteVal = (document.getElementById("filtroCliente")?.value || '').trim();
				if (clienteVal) {
					params.append('busca', clienteVal);
				}
				const dataInicioVal = document.getElementById("filtroDataInicio")?.value;
				if (dataInicioVal) {
					params.append('dataInicio', dataInicioVal + 'T00:00:00');
				}
				const dataFimVal = document.getElementById("filtroDataFim")?.value;
				if (dataFimVal) {
					params.append('dataFim', dataFimVal + 'T23:59:59');
				}

				const queryString = params.toString();
				const url = '/api/v1/admin/pedidos' + (queryString ? '?' + queryString : '');

				const response = await fetch(url);
				if (!response.ok) {
					console.warn("Falha ao obter lista de pedidos via API:", response.status);
					return;
				}

				const data = await response.json();
				const pedidos = Array.isArray(data) ? data : (data.content || []);
				pedidosCache = pedidos;
				renderizarTabelaPedidos(pedidos);
				atualizarMetricas(pedidos);
			} catch (err) {
				console.error("Erro na atualização de pedidos:", err);
			} finally {
				if (manual && refreshIcon) {
					setTimeout(() => {
						refreshIcon.style.animation = "none";
					}, 500);
				}
			}
		}

		function atualizarMetricas(pedidos) {
			const total = pedidos.length;
			let pagos = 0;
			let pendentes = 0;
			let cancelados = 0;
			let rejeitados = 0;
			let faturamento = 0;

			pedidos.forEach(p => {
				const status = (p.status || '').toUpperCase();
				if (status === 'PAGO' || status === 'APROVADO') {
					pagos++;
					faturamento += Number(p.valorTotal || 0);
				} else if (status === 'PENDENTE' || status === 'AGUARDANDO_PAGAMENTO' || status === 'EM_PROCESSAMENTO') {
					pendentes++;
				} else if (status === 'CANCELADO') {
					cancelados++;
				} else {
					rejeitados++;
				}
			});

			const elTotal = document.getElementById("metricaTotalPedidos");
			const elPagos = document.getElementById("metricaPagos");
			const elPend = document.getElementById("metricaPendentes");
			const elFat = document.getElementById("metricaFaturamento");
			const elCountTodos = document.getElementById("countTodos");
			const elCountPago = document.getElementById("countPago");
			const elCountPendente = document.getElementById("countPendente");
			const elCountCancelado = document.getElementById("countCancelado");
			const elCountRejeitado = document.getElementById("countRejeitado");

			if (elTotal) elTotal.innerText = total;
			if (elPagos) elPagos.innerText = pagos;
			if (elPend) elPend.innerText = pendentes;
			if (elFat) elFat.innerText = formatarMoeda(faturamento);

			if (elCountTodos) elCountTodos.innerText = total;
			if (elCountPago) elCountPago.innerText = pagos;
			if (elCountPendente) elCountPendente.innerText = pendentes;
			if (elCountCancelado) elCountCancelado.innerText = cancelados;
			if (elCountRejeitado) elCountRejeitado.innerText = rejeitados;
		}

		function copiarUuid(btn) {
			const uuid = btn.getAttribute("data-uuid");
			if (!uuid) return;
			navigator.clipboard.writeText(uuid).then(() => {
				const orig = btn.innerText;
				btn.innerText = "✓";
				setTimeout(() => { btn.innerText = orig; }, 2000);
			});
		}

		function renderizarTabelaRankingProdutos(produtos, faturamentoTotal) {
			const tbody = document.getElementById("tabelaRankingProdutos");
			if (!tbody) return;

			if (!produtos || produtos.length === 0) {
				tbody.innerHTML = `<tr><td colspan="5" style="text-align: center; color: #888; padding: 30px;">Nenhum produto com registro de vendas no período.</td></tr>`;
				return;
			}

			const maxQtd = Math.max(...produtos.map(p => Number(p.quantidadeVendida || 0)), 1);
			const totalGeral = Number(faturamentoTotal || 0);

			tbody.innerHTML = produtos.map((p, index) => {
				const qtd = Number(p.quantidadeVendida || 0);
				const fat = Number(p.totalFaturado || 0);
				const percQtd = Math.round((qtd / maxQtd) * 100);
				const percFat = totalGeral > 0 ? ((fat / totalGeral) * 100).toFixed(1) : 0;
				const medalha = index === 0 ? '🥇 #1' : (index === 1 ? '🥈 #2' : (index === 2 ? '🥉 #3' : `#${index + 1}`));

				return `
					<tr>
						<td style="font-weight: 700; color: #3da9ff; font-family: 'JetBrains Mono', monospace;">${medalha}</td>
						<td style="font-weight: 700; color: #fff;">${p.nome || '-'}</td>
						<td>
							<div style="display: flex; align-items: center; gap: 8px;">
								<span style="font-weight: 700; color: #00e676; min-width: 30px;">${qtd} un.</span>
								<div class="progress-bar-bg" style="max-width: 120px;">
									<div class="progress-bar-fill" style="width: ${percQtd}%;"></div>
								</div>
							</div>
						</td>
						<td style="font-weight: 700; color: #00e676; font-family: 'JetBrains Mono', monospace;">
							${formatarMoeda(fat)}
						</td>
						<td>
							<span style="font-size: 11px; padding: 3px 8px; border-radius: 4px; background: rgba(0, 153, 255, 0.12); color: #3da9ff; font-weight: 700;">
								${percFat}% do faturamento
							</span>
						</td>
					</tr>
				`;
			}).join('');
		}

		function renderizarGraficoEvolucao(dadosPeriodo) {
			const canvas = document.getElementById("graficoEvolucaoVendas");
			if (!canvas || typeof Chart === 'undefined') return;

			const labels = (dadosPeriodo || []).map(d => d.periodo);
			const valoresFaturamento = (dadosPeriodo || []).map(d => Number(d.totalFaturado || 0));
			const qtdPedidos = (dadosPeriodo || []).map(d => Number(d.quantidadePedidos || 0));

			if (chartInstance) {
				chartInstance.destroy();
			}

			const ctx = canvas.getContext("2d");
			const gradient = ctx.createLinearGradient(0, 0, 0, 300);
			gradient.addColorStop(0, 'rgba(0, 230, 118, 0.35)');
			gradient.addColorStop(1, 'rgba(0, 230, 118, 0.0)');

					labels: labels,
					datasets: [

						mode: 'index',

									if (context.dataset.yAxisID === 'y') {
										return 'Faturamento: ' + formatarMoeda(context.parsed.y);
									}
									return 'Vendas: ' + context.parsed.y + ' pedido(s)';
								}
							}
						}
					},
					scales: {
						x: {
							grid: {
								color: 'rgba(255, 255, 255, 0.05)'
							},
							ticks: {
